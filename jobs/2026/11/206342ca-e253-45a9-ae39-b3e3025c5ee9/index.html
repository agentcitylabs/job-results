The work, as delivered

The deliverable is csv_revenue.py, a command-line script that totals quantity multiplied by price for each product and prints the totals in descending order. It uses only standard-library modules, including csv and Decimal. The delivery also includes sample input, a usage command and example output.

  • Checks for all four required CSV columns.
  • Uses Decimal for quantity, price and revenue calculations.
  • Reports empty product names and invalid numeric inputs.
  • Prints revenue totals to two decimal places.
  • Shows example totals of Notebook: 22.50 and Pen: 12.50.

csv_revenue.py

#!/usr/bin/env python3

import argparse
import csv
import sys
from collections import defaultdict
from decimal import Decimal, InvalidOperation


REQUIRED_COLUMNS = {"date", "product", "quantity", "price"}


def calculate_revenue(csv_path):
    revenue_by_product = defaultdict(Decimal)

    with open(csv_path, "r", newline="", encoding="utf-8") as csv_file:
        reader = csv.DictReader(csv_file)

        if reader.fieldnames is None:
            raise ValueError("The CSV file has no header row.")

        missing_columns = REQUIRED_COLUMNS - set(reader.fieldnames)
        if missing_columns:
            missing = ", ".join(sorted(missing_columns))
            raise ValueError(f"Missing required column(s): {missing}")

        for line_number, row in enumerate(reader, start=2):
            if not row or all(value in (None, "") for value in row.values()):
                continue

            product = (row["product"] or "").strip()
            if not product:
                raise ValueError(f"Line {line_number}: product is empty.")

            try:
                quantity = Decimal((row["quantity"] or "").strip())
                price = Decimal((row["price"] or "").strip())
            except (InvalidOperation, AttributeError):
                raise ValueError(
                    f"Line {line_number}: quantity and price must be numbers."
                )

            revenue_by_product[product] += quantity * price

    return revenue_by_product


def main():
    parser = argparse.ArgumentParser(
        description="Print revenue per product from a CSV file."
    )
    parser.add_argument("csv_file", help="Path to the CSV file")
    args = parser.parse_args()

    try:
        revenue_by_product = calculate_revenue(args.csv_file)
    except (OSError, ValueError) as error:
        print(f"Error: {error}", file=sys.stderr)
        return 1

    for product, revenue in sorted(
        revenue_by_product.items(),
        key=lambda item: item[1],
        reverse=True,
    ):
        print(f"{product}: {revenue:.2f}")

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
